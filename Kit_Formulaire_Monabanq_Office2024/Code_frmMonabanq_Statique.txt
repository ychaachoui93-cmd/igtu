Private Sub UserForm_Initialize()
    Dim initErr As Long, initDesc As String
    On Error GoTo InitEchec
    mInitStep = "configuration visuelle"
    ConfigurerInterface
    mInitStep = "chargement du logo"
    ChargerLogo
    mInitStep = "identifiant et horodatage"
    Me.Controls("txtPersoID").Value = Environ$("USERNAME")
    Me.Controls("txtDateHeure").Value = Format$(Now, "dd/mm/yyyy hh:nn:ss")
    ThisWorkbook.Worksheets("Dashboard").Range("B2").Value = Environ$("USERNAME")
    mLoading = True
    mInitStep = "chargement des activites"
    ChargerTaches
    mLoading = False
    mInitStep = "mise a jour du recapitulatif"
    TaskChanged
    mLoading = True
    ChargerRecap
    mLoading = False
    mInitStep = "initialisation terminee"
    Exit Sub
InitEchec:
    initErr = Err.Number
    initDesc = Err.Description
    mInitFailed = True
    mLoading = False
    mInitDiagnostic = "Etape : " & mInitStep & vbCrLf & _
                      "Erreur " & initErr & " : " & initDesc
End Sub

Public Property Get InitialisationOK() As Boolean
    InitialisationOK = Not mInitFailed

    DiagnosticInitialisation = mInitDiagnostic
End Property

Private Sub ChargerGraphique()
    Dim cfg As Worksheet, wt As Worksheet, r As Long, last As Long
    Dim task1 As String, task2 As String, task As String, state As String
    Dim n1 As Long, n2 As Long, maxN As Long, width1 As Single, width2 As Single
    Set cfg = ThisWorkbook.Worksheets(SH_CFG)
    Set wt = ThisWorkbook.Worksheets(SH_TEMP)
    task1 = Trim$(CStr(cfg.Range("H2").Value))
    task2 = Trim$(CStr(cfg.Range("H3").Value))
    If task1 = "" Then task1 = "Activite 1"
    If task2 = "" Then task2 = "Activite 2"
    last = wt.Cells(wt.Rows.Count, 1).End(xlUp).Row
    For r = 2 To last
        If Len(wt.Cells(r, 1).Value) > 0 And IsDate(wt.Cells(r, 4).Value) Then
            If DateValue(CDate(wt.Cells(r, 4).Value)) = Date Then
                state = CStr(wt.Cells(r, 8).Value)
                If state <> EtatSupprimee() Then
                    task = Trim$(CStr(wt.Cells(r, 5).Value))
                    If StrComp(task, task1, vbTextCompare) = 0 Then n1 = n1 + 1
                    If StrComp(task, task2, vbTextCompare) = 0 Then n2 = n2 + 1
                End If
            End If
        End If
    Next r
    Me.Controls("lblGraphTask1").Caption = task1
    Me.Controls("lblGraphTask2").Caption = task2
    Me.Controls("lblGraphValue1").Caption = CStr(n1)
    Me.Controls("lblGraphValue2").Caption = CStr(n2)
    maxN = n1
    If n2 > maxN Then maxN = n2
    width1 = 1: width2 = 1
    If maxN > 0 Then
        width1 = Application.Max(1, 330 * n1 / maxN)
        width2 = Application.Max(1, 330 * n2 / maxN)
    End If
    With Me.Controls("lblGraphBar1")
        .Width = width1: .Visible = (n1 > 0): .ZOrder 0
    End With
    With Me.Controls("lblGraphBar2")
        .Width = width2: .Visible = (n2 > 0): .ZOrder 0
    End With
End Sub

Private Sub ChargerTaches()
    Dim cfg As Worksheet, c As MSForms.ComboBox, r As Long, v As String
    Set cfg = ThisWorkbook.Worksheets(SH_CFG)
    Set c = Me.Controls("cboTache")
    c.Clear
    For r = 2 To 3
        v = Trim$(CStr(cfg.Cells(r, 8).Value))
        If v <> "" Then c.AddItem v
    Next r
    c.ListIndex = -1
End Sub

Public Sub TaskChanged()
    Dim cfg As Worksheet, task As String, previous As String, col As Long, r As Long
    Dim c As MSForms.ComboBox, v As String
    If mLoading Then Exit Sub
    Set cfg = ThisWorkbook.Worksheets(SH_CFG)
    Set c = Me.Controls("cboTypologie")
    previous = CStr(c.Value)
    c.Clear
    task = Trim$(CStr(Me.Controls("cboTache").Value))
    If task = Trim$(CStr(cfg.Range("H2").Value)) Then
        col = 9
    ElseIf task = Trim$(CStr(cfg.Range("H3").Value)) Then
        col = 10
    Else
        Exit Sub
    End If
    For r = 2 To 7
        v = Trim$(CStr(cfg.Cells(r, col).Value))
        If v <> "" Then c.AddItem v
    Next r
    For r = 0 To c.ListCount - 1
        If StrComp(CStr(c.List(r)), previous, vbTextCompare) = 0 Then
            c.ListIndex = r
            Exit Sub
        End If
    Next r
    c.ListIndex = -1
End Sub

Private Sub ChargerRecap()
    Dim wt As Worksheet, r As Long, last As Long, id As Long, etat As String, dateVal As Date
    Dim lb As MSForms.ListBox, addedToday As Long, deletedToday As Long, savedToday As Long, pendingAll As Long
    Set wt = ThisWorkbook.Worksheets(SH_TEMP)
    Set lb = Me.Controls("lstRecap")
    lb.Clear
    last = wt.Cells(wt.Rows.Count, 1).End(xlUp).Row
    For r = 2 To last
        If IsNumeric(wt.Cells(r, 1).Value) And IsDate(wt.Cells(r, 4).Value) Then
            id = CLng(wt.Cells(r, 1).Value)
            etat = CStr(wt.Cells(r, 8).Value)
            dateVal = CDate(wt.Cells(r, 4).Value)
            If etat <> EtatSupprimee() And (DateValue(dateVal) = Date Or etat = EtatAjoutee()) Then
                lb.AddItem CStr(id)
                lb.List(lb.ListCount - 1, 1) = CStr(wt.Cells(r, 3).Value)
                lb.List(lb.ListCount - 1, 2) = Format$(dateVal, "dd/mm/yyyy hh:nn")
                lb.List(lb.ListCount - 1, 3) = CStr(wt.Cells(r, 5).Value)
                lb.List(lb.ListCount - 1, 4) = CStr(wt.Cells(r, 6).Value)
                lb.List(lb.ListCount - 1, 5) = etat
                lb.List(lb.ListCount - 1, 6) = CStr(wt.Cells(r, 7).Value)
                lb.List(lb.ListCount - 1, 7) = CStr(wt.Cells(r, 2).Value)
            End If
            If DateValue(dateVal) = Date Then
                If etat = EtatAjoutee() Then addedToday = addedToday + 1
                If etat = EtatSupprimee() Then deletedToday = deletedToday + 1
                If etat = EtatEnregistree() Then savedToday = savedToday + 1
            End If
            If etat = EtatAjoutee() Then pendingAll = pendingAll + 1
        End If
    Next r
    Me.Controls("lblResume").Caption = "REALISATIONS DU " & UCase$(Format$(Date, "dd mmmm yyyy"))
    Me.Controls("lblCarteAjouteesValeur").Caption = CStr(addedToday)
    Me.Controls("lblCarteEnregistreesValeur").Caption = CStr(savedToday)
    Me.Controls("lblCarteSupprimeesValeur").Caption = CStr(deletedToday)
    Me.Controls("lblCarteAttenteValeur").Caption = CStr(pendingAll)
    ChargerGraphique
End Sub

Public Sub SelectionChanged()
    Dim lb As MSForms.ListBox, id As Long, r As Long, wt As Worksheet
    If mLoading Then Exit Sub
    Set lb = Me.Controls("lstRecap")
    If lb.ListIndex < 0 Then Exit Sub
    id = CLng(lb.List(lb.ListIndex, 0))
    Set wt = ThisWorkbook.Worksheets(SH_TEMP)
    r = TrouverID(wt, id)
    If r = 0 Then Exit Sub
    mSelectedID = id
    Me.Controls("txtDossier").Value = CStr(wt.Cells(r, 3).Value)
    Me.Controls("txtDateHeure").Value = Format$(CDate(wt.Cells(r, 4).Value), "dd/mm/yyyy hh:nn:ss")
    Me.Controls("cboTache").Value = CStr(wt.Cells(r, 5).Value)
    Me.Controls("cboTypologie").Value = CStr(wt.Cells(r, 6).Value)
    Me.Controls("txtCommentaire").Value = CStr(wt.Cells(r, 7).Value)
End Sub

Public Sub Ajouter()
    Dim agent As String, dossier As String, task As String, typ As String, comment As String
    Dim stamp As Date, wt As Worksheet, r As Long, id As Long
    agent = Environ$("USERNAME")
    dossier = Trim$(CStr(Me.Controls("txtDossier").Value))
    task = Trim$(CStr(Me.Controls("cboTache").Value))
    typ = Trim$(CStr(Me.Controls("cboTypologie").Value))
    comment = CStr(Me.Controls("txtCommentaire").Value)
    If dossier = "" Or task = "" Or typ = "" Or Trim$(comment) = "" Then
        MsgBox "Renseignez le numero de dossier, la tache, la typologie et le commentaire obligatoire.", vbExclamation, "Saisie incomplete"
        Exit Sub
    End If
    If Not TypologieValide(task, typ) Then
        MsgBox "Cette typologie ne correspond pas a la tache selectionnee.", vbExclamation, "Typologie invalide"
        Exit Sub
    End If
    stamp = Now
    If Doublon(agent, dossier, stamp, task, typ, comment, 0) Then
        MsgBox "Cette ligne existe deja pour cet agent aujourd'hui (commentaire compris).", vbExclamation, "Doublon"
        Exit Sub
    End If
    If MsgBox("Confirmer l'ajout de cette ligne ?", vbYesNo + vbQuestion, "Ajouter une ligne") <> vbYes Then Exit Sub
    Set wt = ThisWorkbook.Worksheets(SH_TEMP)
    id = ProchainID(wt)
    r = ProchaineLigne(wt, 1)
    wt.Cells(r, 1).Resize(1, 8).Value = Array(id, agent, dossier, stamp, task, typ, comment, EtatAjoutee())
    wt.Cells(r, 4).NumberFormat = "dd/mm/yyyy hh:mm:ss"
    ThisWorkbook.Save
    ViderChamps
    ChargerRecap
    MsgBox "La ligne " & id & " est ajoutee et gardee dans la file d'attente.", vbInformation, "Monabanq"
End Sub

Public Sub Modifier()
    Dim wt As Worksheet, r As Long, dossier As String, task As String, typ As String, comment As String
    If mSelectedID = 0 Then
        MsgBox "Selectionnez d'abord une ligne dans le recapitulatif.", vbExclamation, "Aucune ligne selectionnee"
        Exit Sub
    End If
    Set wt = ThisWorkbook.Worksheets(SH_TEMP)
    r = TrouverID(wt, mSelectedID)
    If r = 0 Then
        MsgBox "La ligne selectionnee est introuvable.", vbExclamation, "Modification impossible"
        Exit Sub
    End If
    If CStr(wt.Cells(r, 8).Value) <> EtatAjoutee() Then
        MsgBox "Seules les lignes Ajoutees non enregistrees peuvent etre modifiees.", vbExclamation, "Modification impossible"
        Exit Sub
    End If
    dossier = Trim$(CStr(Me.Controls("txtDossier").Value))
    task = Trim$(CStr(Me.Controls("cboTache").Value))
    typ = Trim$(CStr(Me.Controls("cboTypologie").Value))
    comment = CStr(Me.Controls("txtCommentaire").Value)
    If dossier = "" Or task = "" Or typ = "" Or Trim$(comment) = "" Or Not TypologieValide(task, typ) Then
        MsgBox "Verifiez le numero de dossier, la tache, la typologie et le commentaire obligatoire.", vbExclamation, "Saisie invalide"
        Exit Sub
    End If
    If Doublon(CStr(wt.Cells(r, 2).Value), dossier, CDate(wt.Cells(r, 4).Value), task, typ, comment, mSelectedID) Then
        MsgBox "Cette modification creerait un doublon (commentaire compris).", vbExclamation, "Doublon"
        Exit Sub
    End If
    If MsgBox("Confirmer la modification de la ligne " & mSelectedID & " ?", vbYesNo + vbQuestion, "Modifier une ligne") <> vbYes Then Exit Sub
    wt.Cells(r, 3).Value = dossier
    wt.Cells(r, 5).Value = task
    wt.Cells(r, 6).Value = typ
    wt.Cells(r, 7).Value = comment
    ThisWorkbook.Save
    ChargerRecap
    MsgBox "La ligne " & mSelectedID & " a ete modifiee.", vbInformation, "Monabanq"
End Sub

Public Sub Supprimer()
    Dim wt As Worksheet, r As Long, id As Long
    If mSelectedID = 0 Then
        MsgBox "Selectionnez d'abord une ligne dans le recapitulatif.", vbExclamation, "Aucune ligne selectionnee"
        Exit Sub
    End If
    id = mSelectedID
    Set wt = ThisWorkbook.Worksheets(SH_TEMP)
    r = TrouverID(wt, id)
    If r = 0 Then
        MsgBox "La ligne selectionnee est introuvable.", vbExclamation, "Suppression impossible"
        Exit Sub
    End If
    If CStr(wt.Cells(r, 8).Value) <> EtatAjoutee() Then
        MsgBox "Seules les lignes Ajoutees non enregistrees peuvent etre supprimees.", vbExclamation, "Suppression impossible"
        Exit Sub
    End If
    If MsgBox("Confirmer la suppression de la ligne " & id & " ?" & vbCrLf & "Elle sera retiree du recapitulatif.", vbYesNo + vbQuestion, "Supprimer une ligne") <> vbYes Then Exit Sub
    wt.Cells(r, 8).Value = EtatSupprimee()
    ThisWorkbook.Save
    mSelectedID = 0
    ViderChamps
    ChargerRecap
    MsgBox "La ligne " & id & " est supprimee du recapitulatif.", vbInformation, "Monabanq"
End Sub

Public Sub Enregistrer()
    Dim wt As Worksheet, wh As Worksheet
    Dim r As Long, last As Long, id As Long, total As Long, savedCount As Long
    Dim agent As String, csvPath As String, stamp As Date
    Dim errNum As Long, errDesc As String

    Set wt = ThisWorkbook.Worksheets(SH_TEMP)
    last = wt.Cells(wt.Rows.Count, 1).End(xlUp).Row
    For r = 2 To last
        If CStr(wt.Cells(r, 8).Value) = EtatAjoutee() Then total = total + 1
    Next r
    If total = 0 Then
        MsgBox "Aucune ligne ajoutee n'attend d'enregistrement.", vbInformation, "Monabanq"
        Exit Sub
    End If
    If MsgBox("Enregistrer les " & total & " ligne(s) ajoutee(s) dans les CSV du dossier defini en Configuration!B9 ?", _
              vbYesNo + vbQuestion, "Confirmer l'enregistrement") <> vbYes Then Exit Sub

    On Error GoTo GestionErreur
    Set wh = ThisWorkbook.Worksheets(SH_HIST)
    For r = 2 To last
        If CStr(wt.Cells(r, 8).Value) = EtatAjoutee() Then
            id = CLng(wt.Cells(r, 1).Value)
            agent = CStr(wt.Cells(r, 2).Value)
            stamp = CDate(wt.Cells(r, 4).Value)
            csvPath = CheminCsv(agent, stamp)
            AppendCSV csvPath, wt, r
            AjouterHistorique wh, wt, r
            wt.Cells(r, 8).Value = EtatEnregistree()
            savedCount = savedCount + 1
        End If
    Next r

    ThisWorkbook.Save
    mSelectedID = 0
    ViderChamps
    ChargerRecap
    MsgBox savedCount & " ligne(s) enregistree(s) dans le(s) CSV du dossier defini en Configuration!B9 et ajoutee(s) a Historique.", _
           vbInformation, "Monabanq"
    Exit Sub

GestionErreur:
    errNum = Err.Number
    errDesc = Err.Description
    On Error Resume Next
    ThisWorkbook.Save
    mSelectedID = 0
    ViderChamps
    ChargerRecap
    On Error GoTo 0
    MsgBox savedCount & " ligne(s) sur " & total & " enregistree(s). Echec sur la ligne " & id & ": " & _
           errNum & " - " & errDesc & vbCrLf & _
           "Les lignes deja enregistrees sont marquees dans DATA_Temp; les autres restent en attente.", _
           vbCritical, "Erreur CSV"
End Sub

Public Sub Dashboard()
    On Error GoTo GestionErreur
    ThisWorkbook.Activate
    With ThisWorkbook.Worksheets("Dashboard")
        .Visible = xlSheetVisible
        .Activate
    End With
    Application.Calculate
    Exit Sub
GestionErreur:
    MsgBox "Impossible d'ouvrir la feuille Dashboard : " & Err.Description, vbExclamation, "Monabanq"
End Sub

Public Sub Fermer()
    Dim nb As Long, ans As VbMsgBoxResult
    nb = NombreEnAttente
    If nb > 0 Then
        ans = MsgBox(nb & " ligne(s) Ajoutee(s) ne sont pas encore dans le CSV." & vbCrLf & "Voulez-vous enregistrer le classeur pour conserver le cache et fermer le formulaire ?", vbYesNo + vbQuestion, "Fermer Monabanq")
        If ans <> vbYes Then Exit Sub
        ThisWorkbook.Save
    Else
        If MsgBox("Fermer le formulaire Monabanq ?", vbYesNo + vbQuestion, "Fermer") <> vbYes Then Exit Sub
    End If
    mAllowClose = True
    Unload Me
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If Not mAllowClose Then
        Cancel = True
        Fermer
    End If
End Sub

Private Function EtatAjoutee() As String

End Function

Private Function EtatSupprimee() As String
    EtatSupprimee = "Supprim" & ChrW(233) & "e"
End Function

Private Function EtatEnregistree() As String
    EtatEnregistree = "Enregistr" & ChrW(233) & "e"

Private Function TypologieValide(ByVal task As String, ByVal typ As String) As Boolean
    Dim cfg As Worksheet, col As Long, r As Long
    Set cfg = ThisWorkbook.Worksheets(SH_CFG)
    If StrComp(task, CStr(cfg.Range("H2").Value), vbTextCompare) = 0 Then
        col = 9
    ElseIf StrComp(task, CStr(cfg.Range("H3").Value), vbTextCompare) = 0 Then
        col = 10
    Else
        Exit Function
    End If
    For r = 2 To 7
        If StrComp(Trim$(CStr(cfg.Cells(r, col).Value)), typ, vbTextCompare) = 0 And Len(typ) > 0 Then
            TypologieValide = True
            Exit Function
        End If
    Next r
End Function

Private Function Doublon(ByVal agent As String, ByVal dossier As String, ByVal jour As Date, ByVal task As String, ByVal typ As String, ByVal comment As String, ByVal ignoreID As Long) As Boolean
    Dim sh As Worksheet, r As Long, last As Long
    Set sh = ThisWorkbook.Worksheets(SH_TEMP)
    last = sh.Cells(sh.Rows.Count, 1).End(xlUp).Row
    For r = 2 To last
        If Val(sh.Cells(r, 1).Value) <> ignoreID And Len(sh.Cells(r, 1).Value) > 0 And IsDate(sh.Cells(r, 4).Value) And CStr(sh.Cells(r, 8).Value) <> EtatSupprimee() Then
            If CStr(sh.Cells(r, 2).Value) = agent And StrComp(CStr(sh.Cells(r, 3).Value), dossier, vbTextCompare) = 0 _
                And DateValue(sh.Cells(r, 4).Value) = DateValue(jour) _
                And StrComp(CStr(sh.Cells(r, 5).Value), task, vbTextCompare) = 0 _
                And StrComp(CStr(sh.Cells(r, 6).Value), typ, vbTextCompare) = 0 _
                And StrComp(Trim$(CStr(sh.Cells(r, 7).Value)), Trim$(comment), vbTextCompare) = 0 Then
                Doublon = True
                Exit Function
            End If
        End If
    Next r
    Set sh = ThisWorkbook.Worksheets(SH_HIST)
    last = sh.Cells(sh.Rows.Count, 1).End(xlUp).Row
    For r = 2 To last
        If Len(sh.Cells(r, 1).Value) > 0 And IsDate(sh.Cells(r, 3).Value) Then
            If StrComp(CStr(sh.Cells(r, 1).Value), agent, vbTextCompare) = 0 _
                And StrComp(CStr(sh.Cells(r, 2).Value), dossier, vbTextCompare) = 0 _
                And DateValue(sh.Cells(r, 3).Value) = DateValue(jour) _
                And StrComp(CStr(sh.Cells(r, 4).Value), task, vbTextCompare) = 0 _
                And StrComp(CStr(sh.Cells(r, 5).Value), typ, vbTextCompare) = 0 _
                And StrComp(Trim$(CStr(sh.Cells(r, 6).Value)), Trim$(comment), vbTextCompare) = 0 Then
                Doublon = True
                Exit Function
            End If
        End If
    Next r
End Function

Private Function ProchainID(ByVal sh As Worksheet) As Long
    Dim last As Long, r As Long, mx As Long
    last = sh.Cells(sh.Rows.Count, 1).End(xlUp).Row
    For r = 2 To last
        If IsNumeric(sh.Cells(r, 1).Value) Then mx = Application.Max(mx, CLng(sh.Cells(r, 1).Value))
    Next r
    ProchainID = mx + 1
End Function

Private Function ProchaineLigne(ByVal sh As Worksheet, ByVal col As Long) As Long
    ProchaineLigne = sh.Cells(sh.Rows.Count, col).End(xlUp).Row + 1
    If ProchaineLigne < 2 Then ProchaineLigne = 2
End Function

Private Function TrouverID(ByVal sh As Worksheet, ByVal id As Long) As Long
    Dim r As Long, last As Long
    last = sh.Cells(sh.Rows.Count, 1).End(xlUp).Row
    For r = 2 To last
        If Val(sh.Cells(r, 1).Value) = id Then
            TrouverID = r
            Exit Function
        End If
    Next r
End Function

Private Sub AjouterHistorique(ByVal wh As Worksheet, ByVal wt As Worksheet, ByVal sourceRow As Long)
    Dim r As Long
    r = ProchaineLigne(wh, 1)
    wh.Cells(r, 1).Resize(1, 6).Value = Array(wt.Cells(sourceRow, 2).Value, wt.Cells(sourceRow, 3).Value, wt.Cells(sourceRow, 4).Value, wt.Cells(sourceRow, 5).Value, wt.Cells(sourceRow, 6).Value, wt.Cells(sourceRow, 7).Value)
    wh.Cells(r, 3).NumberFormat = "dd/mm/yyyy hh:mm:ss"
End Sub

Private Function CheminCsv(ByVal agent As String, ByVal stamp As Date) As String
    Dim base As String, root As String, folder As String, fileName As String

    base = Trim$(CStr(ThisWorkbook.Worksheets(SH_CFG).Range("B9").Value))
    If Len(base) = 0 Then
        Err.Raise vbObjectError + 2120, "Monabanq", _
            "Renseignez Configuration!B9 avec le dossier de destination des fichiers CSV."
    End If

    base = Replace(base, "/", Application.PathSeparator)
    Do While Len(base) > 3 And Right$(base, 1) = Application.PathSeparator
        base = Left$(base, Len(base) - 1)
    Loop

    If (Len(base) >= 3 And Mid$(base, 2, 2) = ":" & Application.PathSeparator) _
       Or Left$(base, 2) = Application.PathSeparator & Application.PathSeparator _
       Or Left$(base, 1) = Application.PathSeparator Then
        root = base
    Else
        If Len(ThisWorkbook.Path) = 0 Then
            Err.Raise vbObjectError + 2122, "Monabanq", _
                "Enregistrez d'abord le classeur .xlsm pour utiliser un dossier relatif."
        End If
        root = ThisWorkbook.Path & Application.PathSeparator & base
    End If

    folder = root & Application.PathSeparator & Format$(stamp, "yyyy") & _
             Application.PathSeparator & Format$(stamp, "mm")
    CreerDossier folder
    fileName = NomFichier(agent) & ".csv"
    CheminCsv = folder & Application.PathSeparator & fileName
End Function

Private Sub AppendCSV(ByVal filePath As String, ByVal wt As Worksheet, ByVal r As Long)
    Dim stm As Object, txt As String, header As String, firstLine As String
    header = "PersoID;n" & ChrW(176) & " de dossier;date/heure de saisie;Tache;Typologie;commentaire" & vbCrLf
    If Dir$(filePath) <> "" Then
        Set stm = CreateObject("ADODB.Stream")
        stm.Type = 2: stm.Charset = "utf-8": stm.Open
        stm.LoadFromFile filePath
        txt = stm.ReadText
        stm.Close
    Else
        txt = header
    End If
    If Len(txt) = 0 Then txt = header
    firstLine = txt
    If InStr(1, txt, vbCrLf) > 0 Then firstLine = Left$(txt, InStr(1, txt, vbCrLf) - 1)
    If InStr(1, firstLine, ";", vbBinaryCompare) = 0 And _
       InStr(1, firstLine, ",", vbBinaryCompare) > 0 Then
        txt = ConvertirCsvPointVirgule(txt)
    End If
    If Right$(txt, 1) <> vbLf Then txt = txt & vbCrLf
    txt = txt & CsvField(wt.Cells(r, 2).Value) & ";" & CsvField(wt.Cells(r, 3).Value) & ";" & CsvField(Format$(CDate(wt.Cells(r, 4).Value), "yyyy-mm-dd hh:nn:ss")) & ";" & CsvField(wt.Cells(r, 5).Value) & ";" & CsvField(wt.Cells(r, 6).Value) & ";" & CsvField(wt.Cells(r, 7).Value) & vbCrLf
    Set stm = CreateObject("ADODB.Stream")
    stm.Type = 2: stm.Charset = "utf-8": stm.Open
    stm.WriteText txt
    stm.SaveToFile filePath, 2
    stm.Close
End Sub

Private Function CsvField(ByVal value As Variant) As String
    CsvField = """" & Replace(CStr(value), """", """""") & """"
End Function

Private Function ConvertirCsvPointVirgule(ByVal source As String) As String
    Dim i As Long, ch As String, insideQuotes As Boolean, result As String
    i = 1
    Do While i <= Len(source)
        ch = Mid$(source, i, 1)
        If ch = """" Then
            result = result & ch
            If insideQuotes And i < Len(source) And Mid$(source, i + 1, 1) = """" Then
                result = result & """"
                i = i + 1
            Else
                insideQuotes = Not insideQuotes
            End If
        ElseIf ch = "," And Not insideQuotes Then
            result = result & ";"
        Else
            result = result & ch
        End If
        i = i + 1
    Loop
    ConvertirCsvPointVirgule = result
End Function

Private Sub CreerDossier(ByVal folderPath As String)
    Dim fso As Object, parent As String
    Set fso = CreateObject("Scripting.FileSystemObject")
    If fso.FolderExists(folderPath) Then Exit Sub
    parent = fso.GetParentFolderName(folderPath)
    If Len(parent) > 0 And Not fso.FolderExists(parent) Then CreerDossier parent
    If Not fso.FolderExists(folderPath) Then fso.CreateFolder folderPath
End Sub

Private Function NomFichier(ByVal value As String) As String
    Dim chars As Variant, i As Long
    chars = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    NomFichier = Trim$(value)
    For i = LBound(chars) To UBound(chars)
        NomFichier = Replace(NomFichier, CStr(chars(i)), "_")
    Next i
    If NomFichier = "" Then NomFichier = "agent"
End Function
